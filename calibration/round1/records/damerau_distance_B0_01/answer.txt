def damerau_distance(arg):
    a, b = arg["a"], arg["b"]
    n, m = len(a), len(b)
    if not n:
        return m
    if not m:
        return n

    limit = n + m
    d = [[0] * (m + 2) for _ in range(n + 2)]
    d[0][0] = limit

    for i in range(n + 1):
        d[i + 1][0] = limit
        d[i + 1][1] = i
    for j in range(m + 1):
        d[0][j + 1] = limit
        d[1][j + 1] = j

    last_seen = {}
    for i in range(1, n + 1):
        last_match = 0
        for j in range(1, m + 1):
            previous_i = last_seen.get(b[j - 1], 0)
            previous_j = last_match
            cost = 1
            if a[i - 1] == b[j - 1]:
                cost = 0
                last_match = j

            d[i + 1][j + 1] = min(
                d[i][j] + cost,
                d[i + 1][j] + 1,
                d[i][j + 1] + 1,
                d[previous_i][previous_j]
                + (i - previous_i - 1)
                + 1
                + (j - previous_j - 1),
            )
        last_seen[a[i - 1]] = i

    return d[n + 1][m + 1]
