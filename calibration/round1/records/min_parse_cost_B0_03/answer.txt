def min_parse_cost(data):
    from heapq import heappop, heappush

    k = data["nonterminals"]
    text = data["text"]
    n = len(text)
    width = n + 1
    chart_size = width * width
    terminal_ids = {"a": k, "b": k + 1}
    count = k + 2

    empty_rules = []
    unary_rules = []
    binary_rules = []

    for lhs, rhs, cost in data["rules"]:
        symbols = [
            value if kind == "N" else terminal_ids[value]
            for kind, value in rhs
        ]
        if not symbols:
            empty_rules.append((lhs, cost))
        elif len(symbols) == 1:
            unary_rules.append((lhs, symbols[0], cost))
        elif len(symbols) == 2:
            binary_rules.append((lhs, symbols[0], symbols[1], cost))
        else:
            auxiliary = count
            count += 1
            binary_rules.append((lhs, symbols[0], auxiliary, cost))
            binary_rules.append((auxiliary, symbols[1], symbols[2], 0))

    unary_parents = [[] for _ in range(count)]
    left_parents = [[] for _ in range(count)]
    right_parents = [[] for _ in range(count)]

    for lhs, child, cost in unary_rules:
        unary_parents[child].append((lhs, cost))
    for lhs, left, right, cost in binary_rules:
        left_parents[left].append((lhs, right, cost))
        right_parents[right].append((lhs, left, cost))

    distances = [[float("inf")] * chart_size for _ in range(count)]
    settled = [bytearray(chart_size) for _ in range(count)]
    by_start = [[[] for _ in range(width)] for _ in range(count)]
    by_end = [[[] for _ in range(width)] for _ in range(count)]
    heap = []

    def offer(symbol, start, end, cost):
        position = start * width + end
        if cost < distances[symbol][position]:
            distances[symbol][position] = cost
            heappush(heap, (cost, symbol, start, end))

    for lhs, cost in empty_rules:
        for position in range(width):
            offer(lhs, position, position, cost)

    for position, character in enumerate(text):
        offer(terminal_ids[character], position, position + 1, 0)

    target = data["start"]

    while heap:
        cost, symbol, start, end = heappop(heap)
        position = start * width + end
        if settled[symbol][position] or cost != distances[symbol][position]:
            continue

        settled[symbol][position] = 1
        if symbol == target and start == 0 and end == n:
            return cost

        by_start[symbol][start].append((end, cost))
        by_end[symbol][end].append((start, cost))

        for lhs, rule_cost in unary_parents[symbol]:
            offer(lhs, start, end, cost + rule_cost)

        for lhs, right, rule_cost in left_parents[symbol]:
            for right_end, right_cost in by_start[right][end]:
                offer(lhs, start, right_end, cost + right_cost + rule_cost)

        for lhs, left, rule_cost in right_parents[symbol]:
            for left_start, left_cost in by_end[left][start]:
                offer(lhs, left_start, end, left_cost + cost + rule_cost)

    return None
