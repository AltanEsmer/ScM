def damerau_distance(arg):
    a = arg["a"]
    b = arg["b"]
    m, n = len(a), len(b)
    limit = m + n

    distance = [[0] * (n + 2) for _ in range(m + 2)]
    distance[0][0] = limit

    for i in range(m + 1):
        distance[i + 1][0] = limit
        distance[i + 1][1] = i
    for j in range(n + 1):
        distance[0][j + 1] = limit
        distance[1][j + 1] = j

    last_seen = {}
    for i in range(1, m + 1):
        last_match = 0
        for j in range(1, n + 1):
            previous_i = last_seen.get(b[j - 1], 0)
            previous_j = last_match
            cost = 1
            if a[i - 1] == b[j - 1]:
                cost = 0
                last_match = j

            distance[i + 1][j + 1] = min(
                distance[i][j] + cost,
                distance[i + 1][j] + 1,
                distance[i][j + 1] + 1,
                distance[previous_i][previous_j]
                + (i - previous_i - 1)
                + 1
                + (j - previous_j - 1),
            )
        last_seen[a[i - 1]] = i

    return distance[m + 1][n + 1]
