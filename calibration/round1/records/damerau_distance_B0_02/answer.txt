def damerau_distance(arg):
    a, b = arg["a"], arg["b"]
    n, m = len(a), len(b)
    maximum = n + m
    distance = [[0] * (m + 2) for _ in range(n + 2)]
    distance[0][0] = maximum

    for i in range(n + 1):
        distance[i + 1][0] = maximum
        distance[i + 1][1] = i
    for j in range(m + 1):
        distance[0][j + 1] = maximum
        distance[1][j + 1] = j

    last_seen = {}
    for i in range(1, n + 1):
        last_match = 0
        for j in range(1, m + 1):
            previous_i = last_seen.get(b[j - 1], 0)
            previous_j = last_match
            cost = 1
            if a[i - 1] == b[j - 1]:
                cost = 0
                last_match = j

            distance[i + 1][j + 1] = min(
                distance[i][j] + cost,
                distance[i + 1][j] + 1,
                distance[i][j + 1] + 1,
                distance[previous_i][previous_j]
                + (i - previous_i - 1)
                + 1
                + (j - previous_j - 1),
            )
        last_seen[a[i - 1]] = i

    return distance[n + 1][m + 1]
