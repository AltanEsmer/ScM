def pickup_delivery_cost(arg):
    travel_time = arg["time"]
    travel_cost = arg["cost"]
    requests = arg["requests"]
    vehicle = arg["vehicle"]
    capacity = vehicle["capacity"]
    deadline = vehicle["deadline"]
    end = vehicle["end"]
    all_delivered = (1 << len(requests)) - 1
    best = None

    def search(node, now, load, picked, delivered, total_cost):
        nonlocal best

        if now > deadline or (best is not None and total_cost >= best):
            return

        if delivered == all_delivered:
            if now + travel_time[node][end] <= deadline:
                candidate = total_cost + travel_cost[node][end]
                if best is None or candidate < best:
                    best = candidate
            return

        for i, request in enumerate(requests):
            bit = 1 << i
            if delivered & bit:
                continue

            if picked & bit:
                target = request["drop"]
                earliest, latest = request["dw"]
                duration = request["ds"]
                next_load = load - request["load"]
                next_picked = picked
                next_delivered = delivered | bit
            else:
                next_load = load + request["load"]
                if next_load > capacity:
                    continue
                target = request["pick"]
                earliest, latest = request["pw"]
                duration = request["ps"]
                next_picked = picked | bit
                next_delivered = delivered

            service_start = max(now + travel_time[node][target], earliest)
            if service_start > latest:
                continue

            search(
                target,
                service_start + duration,
                next_load,
                next_picked,
                next_delivered,
                total_cost + travel_cost[node][target],
            )

    search(vehicle["start"], vehicle["release"], 0, 0, 0, 0)
    return best
