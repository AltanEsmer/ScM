def best_itinerary(data):
    flights = data["flights"]
    start = data["start"]
    goal = data["goal"]
    if start == goal:
        return []

    best = [None] * len(flights)
    answer = None

    for i in sorted(range(len(flights)), key=lambda i: flights[i][3]):
        origin, destination, departure, arrival = flights[i]
        route = (i,) if origin == start and departure >= data["time"] else None

        for j, previous in enumerate(best):
            if (
                previous is not None
                and flights[j][1] == origin
                and flights[j][3] <= departure
            ):
                candidate = previous + (i,)
                if route is None or (len(candidate), candidate) < (len(route), route):
                    route = candidate

        best[i] = route
        if destination == goal and route is not None:
            candidate = (arrival, len(route), route)
            if answer is None or candidate < answer:
                answer = candidate

    return None if answer is None else list(answer[2])
