def best_itinerary(data):
    flights = data["flights"]
    start = data["start"]
    goal = data["goal"]
    ready = data["time"]

    if start == goal:
        return []

    order = sorted(range(len(flights)), key=lambda i: flights[i][2])
    paths = {}
    best = None

    for i in order:
        origin, destination, departure, arrival = flights[i]
        path = (i,) if origin == start and departure >= ready else None

        for j, previous_path in paths.items():
            if flights[j][1] == origin and flights[j][3] <= departure:
                candidate = previous_path + (i,)
                if path is None or (len(candidate), candidate) < (len(path), path):
                    path = candidate

        if path is None:
            continue

        paths[i] = path
        if destination == goal:
            candidate = (arrival, len(path), path)
            if best is None or candidate < best:
                best = candidate

    return None if best is None else list(best[2])
