def select_jobs(data):
    best = []
    best_profit = 0
    n = len(data)

    for mask in range(1, 1 << n):
        indices = [i for i in range(n) if mask & (1 << i)]
        intervals = sorted((data[i][0], data[i][1]) for i in indices)
        if any(intervals[j - 1][1] > intervals[j][0]
               for j in range(1, len(intervals))):
            continue

        profit = sum(data[i][2] for i in indices)
        if (-profit, len(indices), indices) < (-best_profit, len(best), best):
            best = indices
            best_profit = profit

    return best
