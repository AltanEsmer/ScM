def select_jobs(data):
    best_indices = []
    best_key = (0, 0, [])

    for mask in range(1, 1 << len(data)):
        indices = [i for i in range(len(data)) if mask & (1 << i)]
        intervals = sorted((data[i][0], data[i][1]) for i in indices)

        if any(
            intervals[i - 1][1] > intervals[i][0]
            for i in range(1, len(intervals))
        ):
            continue

        profit = sum(data[i][2] for i in indices)
        key = (-profit, len(indices), indices)
        if key < best_key:
            best_key = key
            best_indices = indices

    return best_indices
