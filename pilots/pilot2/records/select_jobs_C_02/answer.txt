def select_jobs(data):
    n = len(data)
    conflicts = [0] * n
    for i in range(n):
        for j in range(i):
            if data[i][0] < data[j][1] and data[j][0] < data[i][1]:
                conflicts[i] |= 1 << j
                conflicts[j] |= 1 << i

    best = []
    best_profit = 0

    def search(i, mask, profit, selected):
        nonlocal best, best_profit
        if i == n:
            if (
                profit > best_profit
                or (
                    profit == best_profit
                    and (len(selected), selected) < (len(best), best)
                )
            ):
                best_profit = profit
                best = selected.copy()
            return

        search(i + 1, mask, profit, selected)
        if not (conflicts[i] & mask):
            selected.append(i)
            search(i + 1, mask | (1 << i), profit + data[i][2], selected)
            selected.pop()

    search(0, 0, 0, [])
    return best
