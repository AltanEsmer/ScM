from itertools import combinations


def select_jobs(data):
    best_indices = []
    best_profit = 0

    for count in range(1, len(data) + 1):
        for indices in combinations(range(len(data)), count):
            ordered = sorted(indices, key=lambda i: data[i][0])
            if any(
                data[left][1] > data[right][0]
                for left, right in zip(ordered, ordered[1:])
            ):
                continue

            profit = sum(data[i][2] for i in indices)
            candidate = list(indices)
            if profit > best_profit or (
                profit == best_profit
                and (count, candidate) < (len(best_indices), best_indices)
            ):
                best_profit = profit
                best_indices = candidate

    return best_indices
