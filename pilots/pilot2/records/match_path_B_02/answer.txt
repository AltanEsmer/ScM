def match_path(data):
    pattern = data["pattern"]
    text = data["text"]
    tokens = []
    i = 0

    while i < len(pattern):
        char = pattern[i]
        if char == "\\":
            if i + 1 < len(pattern):
                tokens.append(("literal", pattern[i + 1]))
                i += 2
            else:
                tokens.append(("literal", "\\"))
                i += 1
        elif char == "*":
            end = i + 1
            while end < len(pattern) and pattern[end] == "*":
                end += 1
            tokens.append(("globstar" if end - i >= 2 else "star", None))
            i = end
        elif char == "?":
            tokens.append(("question", None))
            i += 1
        else:
            tokens.append(("literal", char))
            i += 1

    previous = [True] + [False] * len(text)
    for kind, value in tokens:
        current = [False] * (len(text) + 1)
        if kind in ("star", "globstar"):
            current[0] = previous[0]
            for j, char in enumerate(text, 1):
                current[j] = previous[j] or (
                    current[j - 1] and (kind == "globstar" or char != "/")
                )
        else:
            for j, char in enumerate(text, 1):
                current[j] = previous[j - 1] and (
                    char != "/" if kind == "question" else char == value
                )
        previous = current

    return previous[-1]
